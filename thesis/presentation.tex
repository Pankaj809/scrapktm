\section{Introduction}
\begin{frame}{Introduction: What is RAG?}
  \begin{block}{\ragterm{Retrieval-Augmented Generation (RAG)} — in plain language}
    Instead of asking an AI to \emph{remember} laws, we first \textbf{search official PDFs} for the right paragraph, then let the AI \textbf{read that paragraph} to answer.
  \end{block}
  \vspace{0.25cm}
  \begin{columns}[T]
    \column{0.48\textwidth}
    \textbf{Why it matters}
    \begin{itemize}
      \item Kathmandu Valley bylaws are in Nepali PDFs
      \item Citizens mix Nepali + English when asking questions
      \item General chatbots lack your local municipal files
    \end{itemize}
    \column{0.48\textwidth}
    \textbf{Our focus}
    \begin{itemize}
      \item Municipal bylaws \& federal laws
      \item Local, offline-friendly retrieval
      \item Measuring when RAG beats direct LLMs
    \end{itemize}
  \end{columns}
  \note{RAG = retrieve first, generate second. Define vectorization later only if asked.}
\end{frame}

%==============================================================================
\section{Problem Statement}
\begin{frame}{Problem Statement}
  \begin{alertblock}{Three barriers for Nepali legal RAG}
    \begin{enumerate}
      \item \textbf{Legacy PDF encoding} — many files use old fonts; extracted text looks like gibberish, not Unicode Nepali.
      \item \textbf{Code-switched queries} — users ask in mixed Nepali/English; English-only embeddings struggle.
      \item \textbf{Evaluation traps} — wrong chunk IDs can make good retrieval look like 0\% accuracy.
    \end{enumerate}
  \end{alertblock}
  \vspace{0.2cm}
  \centering
  \small\textit{Example corruption:}\\
  \texttt{sf7df8f\}F dxfgu/kflnsf} \quad$\rightarrow$\quad \textbf{should be readable Nepali municipal text}
  \note{Stress that 0\% baseline was partly data/encoding, not only ``bad AI.''}
\end{frame}

\section{Methodology}
\begin{frame}{Methodology}
  \begin{columns}[T]
    \column{0.5\textwidth}
    \textbf{Extraction \& chunking}
    \begin{itemize}
      \item \textbf{Router:} text PDFs $\rightarrow$ PyMuPDF; scanned $\rightarrow$ Surya OCR (MPS)
      \item \textbf{Chunking:} heading-aware sections; 512-token fallback
      \item \textbf{Logging:} per-document metrics in JSONL
    \end{itemize}
    \textbf{Tools}
    \begin{itemize}
      \item Python, regex, JSON/JSONL
      \item LangChain Core, scikit-learn
      \item FAISS, sentence-transformers
    \end{itemize}
    \column{0.5\textwidth}
    \textbf{Hybrid retrieval (our fix)}
    \begin{itemize}
      \item \textbf{Metadata aliasing:} inject readable Nepali/English keywords into chunk text
      \item \textbf{Sparse TF-IDF:} word + bigram vectors (keyword matching)
      \item \textbf{Dense baseline:} BGE-M3, MiniLM via FAISS
    \end{itemize}
    \vspace{0.2cm}
    \begin{block}{Terminology note}
      We use \emph{TF-IDF sparse retrieval} in code; BM25 is the same family of keyword methods (not separately implemented).
    \end{block}
  \end{columns}
  \note{Be precise: TF-IDF in local\_langchain\_rag.py; FAISS for dense models in embedding\_benchmark.py.}
\end{frame}

\section{Literature Review}
\begin{frame}{Literature Review (Brief)}
  \begin{itemize}
    \item \textbf{Low-resource NLP:} English-heavy models underperform on Devanagari and code-switching.
    \item \textbf{South Asian digitization:} legacy font encodings break standard OCR and embeddings.
    \item \textbf{Hybrid retrieval:} combining sparse (keyword) and dense (semantic) search improves robustness when text is noisy.
    \item \textbf{Legal IR:} answers often require exact tables, fees, and document lists — retrieval must be precise.
  \end{itemize}

\section{Aim \& Objective}
\begin{frame}{Aim \& Objectives}
  \begin{block}{Primary aim}
    Build and evaluate a \textbf{local RAG pipeline} that retrieves correct clauses from Nepali municipal PDFs using realistic code-switched queries.
  \end{block}
  \textbf{Objectives}
  \begin{enumerate}
    \item Curate and extract a reproducible legal corpus (50 PDFs $\rightarrow$ JSONL chunks).
    \item Measure dense embedding baselines (BGE-M3, MiniLM) on raw extracted text.
    \item Design alias-augmented sparse retrieval and compare Recall@$k$ / MRR.
    \item Contrast \textbf{grounded RAG answers} vs.\ \textbf{direct LLM guesses} on tax and fee queries.
  \end{enumerate}
\end{frame}

\section{Understanding Metrics}
\begin{frame}{Understanding the Metrics}
  \begin{columns}[T]
    \column{0.5\textwidth}
    \begin{block}{Recall@$k$ (we report @3 and @5)}
      Did the \textbf{correct chunk} appear in the top $k$ results?
      \begin{itemize}
        \item 1.0 = found in top-$k$
        \item 0.0 = not found
      \end{itemize}
      \small\textit{Critical for RAG: if retrieval misses, the LLM hallucinates.}
    \end{block}
    \column{0.5\textwidth}
    \begin{block}{MRR (Mean Reciprocal Rank)}
      How high is the correct chunk ranked?
      \begin{itemize}
        \item Rank 1 $\rightarrow$ score 1.0
        \item Rank 2 $\rightarrow$ 0.5
        \item Rank 3 $\rightarrow$ 0.33
      \end{itemize}
      \small\textit{Higher MRR = less noise for the LLM to filter.}
    \end{block}
  \end{columns}
  \note{All numbers on the 50-query leakage-free bank. Sparse and dense scored on identical chunk IDs.}
\end{frame}

\section{Results \& Evaluation}
\begin{frame}{Results: Why Raw Retrieval Fails}
  \textbf{Sparse TF-IDF on \emph{raw} extracted text:}
  \begin{itemize}
    \item Recall@3 = \textbf{0.060}, MRR = \textbf{0.023}, doc-level \textbf{0.140}
    \item Cause: 26/50 documents extract as Preeti font gibberish
    \item Every retriever is broken on raw text --- both dense encoders score $\le$ 0.040
  \end{itemize}
  \vspace{0.2cm}
  \textbf{Key result — deterministic Preeti $\rightarrow$ Unicode repair:} Recall@3 rises to \textbf{0.520} (doc-level \textbf{0.800}) with \emph{no} hand-authored aliases.
  \note{Numbers from paper\_analysis.py on the 50-query bank. Repair is the contribution.}
\end{frame}

\section{Conclusion}
\begin{frame}{Conclusion \& Future Work}
  \textbf{What we showed}
  \begin{itemize}
    \item Built end-to-end pipeline: scrape $\rightarrow$ curate $\rightarrow$ extract $\rightarrow$ chunk $\rightarrow$ retrieve
    \item Raw retrieval fails on corrupted Nepali PDF text (Recall@3 = 0.060, 50-query bank)
    \item \textbf{Deterministic Preeti $\rightarrow$ Unicode repair} lifts Recall@3 to \textbf{0.520} (doc-level 0.800) with no hand-authored aliases
    \item Repair \emph{beats} hand-authored aliasing (0.260) and is necessary-but-not-sufficient for dense retrieval
    \item Local RAG grounds answers in municipal PDFs; direct LLMs do not
  \end{itemize}
  \textbf{Next steps}
  \begin{itemize}
    \item Learned transliteration repair for lossy conjuncts; re-run BGE-M3 fairly
    \item Add BM25 + dense fusion; expand human-labeled query bank
    \item Deploy citizen-facing chatbot with mandatory citations
  \end{itemize}
  \note{Close on impact: accessible legal information for Kathmandu Valley residents.}
\end{frame}
